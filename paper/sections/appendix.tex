\section{Proofs for Section~\ref{sec:theory}}
\label{app:proofs}

Throughout, fix a context $c$ and suppress it when unambiguous. Every
discrepancy considered ($\ell_2$, $\ell_1$, Huber, Cauchy) satisfies
$d(e)\to\infty$ as $|e|\to\infty$, so a policy that assigns zero probability to
a prompt with positive sampling probability has infinite loss. We therefore
restrict attention to full-support $\pi$, for which every log-ratio is finite.

\paragraph{Lemma~\ref{lem:gibbs}.}
Standard; see \citet[App.~A.1]{rafailov2023dpo}. Writing
$\pi^\star \propto \pi_0 e^{r/\tau}$ with normaliser $Z$, the objective equals
$\tau\log Z - \tau\,\mathrm{KL}(\pi\|\pi^\star)$, which Gibbs' inequality
maximizes uniquely at $\pi = \pi^\star$. Taking logarithms,
$r(z) = \tau\log(\pi^\star(z)/\pi_0(z)) + \tau\log Z$. The last term does not
depend on $z$ and cancels in Eq.~\eqref{eq:th-diff}. \qed

\paragraph{Theorem~\ref{thm:consistency}.}
Non-negativity is immediate from $d \ge 0$. Write the residual as
$e(z,z') = [h(z) - h(z')] - [r(z) - r(z')]/s$ with
$h = \beta\log(\pi/\pi_0)$.
($\Leftarrow$) If $\pi = \pi^\star_{\beta s}$, Lemma~\ref{lem:gibbs} with
$\tau = \beta s$ gives $h(z) - h(z') = [r(z) - r(z')]/s$, so every residual is
zero and so is the loss.
($\Rightarrow$) If $\mathcal{L}_{q,s}(\pi) = 0$, then for almost every $c$,
$d(e(z,z')) = 0$ for $q\otimes q$-almost every pair. Since $\mathcal{Z}$ is
finite and $q$ has full support, every pair carries positive mass, so
$e(z,z') = 0$ for \emph{all} $z,z'$. Hence $h(z) - r(z)/s = C$ for some constant
$C$ (depending on $c$ only), i.e.\
$\pi(z) = \pi_0(z)\,e^{r(z)/(\beta s)}\,e^{C/\beta}$. Normalization forces
$e^{C/\beta} = 1/Z$, so $\pi = \pi^\star_{\beta s}$.
Part (ii) follows because a Gibbs policy at temperature $\beta\sigma$ for reward
$r$ is the Gibbs policy at temperature $\beta$ for reward $r/\sigma$, which by
Lemma~\ref{lem:gibbs} maximizes the stated objective. \qed

\paragraph{Lemma~\ref{lem:invariance}.}
For $r' = a r + b$ with $a > 0$ constant within the context:
$r'(z) - r'(z') = a\,[r(z) - r(z')]$ and $\sigma_q(r') = a\,\sigma_q(r)$, so the
standardized difference is unchanged, and so is every residual. The same
algebra holds for the \emph{sample} standard deviation of a finite group, so the
implemented loss is invariant too, up to the $\varepsilon = 10^{-8}$ guard in
its denominator. Without standardization the shift $b$ cancels but the scale
$a$ does not. \qed

\paragraph{Proposition~\ref{prop:trust}.}
Let $\mu = \mathbb{E}_{\pi_0}r$ and $\sigma = \sigma_{\pi_0}$, so $r = \mu + \sigma u$
and $\pi^+ \propto \pi_0 e^{r/(\beta\sigma)} \propto \pi_0 e^{t u}$ with $t = 1/\beta$
(the constant $\mu/(\beta\sigma)$ is absorbed by the normaliser). The
normaliser is $e^{\Lambda_u(t)}$, so
$\mathrm{KL}(\pi^+\|\pi_0) = \mathbb{E}_{\pi^+}[t u] - \Lambda_u(t)$, and the mean
of $u$ under an exponential tilt is $\Lambda_u'(t)$. This gives
Eq.~\eqref{eq:th-trust}; both expressions involve only the law of $u$.
For the constrained characterization, Lemma~\ref{lem:gibbs} (with reward $u$
and temperature $\beta$) gives, for every $\pi$,
$\mathbb{E}_\pi u - \beta\,\mathrm{KL}(\pi\|\pi_0) \le
\mathbb{E}_{\pi^+}u - \beta\,\varepsilon$. So any $\pi$ with
$\mathrm{KL}(\pi\|\pi_0) \le \varepsilon$ has $\mathbb{E}_\pi u \le
\mathbb{E}_{\pi^+}u$, and $\mathbb{E}_\pi r = \mu + \sigma\,\mathbb{E}_\pi u$.
For the expansion, $u$ has mean $0$ and variance $1$, so
$\Lambda_u(t) = t^2/2 + \gamma t^3/6 + O(t^4)$ and
$\Lambda_u'(t) = t + \gamma t^2/2 + O(t^3)$, whence
$t\Lambda_u'(t) - \Lambda_u(t) = t^2/2 + \gamma t^3/3 + O(t^4)$. The plain target
is the tilt $e^{(\sigma/\beta)u}$, i.e.\ the same formulas with
$t = \sigma/\beta$. \qed

\paragraph{Proposition~\ref{prop:mirror}.}
With the reference at $\pi_k$ and the sampler $q = \pi_k$,
Theorem~\ref{thm:consistency}(ii) says the window's target is the maximizer of
$\mathbb{E}_\pi r - \beta\sigma_{\pi_k}\mathrm{KL}(\pi\|\pi_k)$. By
Lemma~\ref{lem:gibbs} this is $\pi_k e^{r/(\beta\sigma_{\pi_k})}$ up to
normalization, which is Eq.~\eqref{eq:th-mirror}. The mirror-ascent step with
the negative-entropy mirror map on the linear objective
$\langle\pi, r\rangle$ is $\arg\max_\pi\, \eta\langle\pi,r\rangle -
\mathrm{KL}(\pi\|\pi_k) = \pi_k e^{\eta r}/Z$, the same update. Unrolling the
product gives $\pi_k \propto \pi_0 e^{T_k r}$. Since $\sigma \le R/2$ with
$R = \max r - \min r$ (Popoviciu), $\eta_j \ge 2/(\beta R)$, so
$T_k \ge 2k/(\beta R) \to \infty$. As $T\to\infty$ the Gibbs policy
$\pi_0 e^{Tr}$ concentrates on $\arg\max r$. \qed

\paragraph{Proposition~\ref{prop:balance}.}
At the window start all log-ratios are zero, so the residual is
$-[r(z) - r(z')]/s$ with $z, z'$ i.i.d.\ from $q$. For $\ell_2$,
$\mathbb{E}[(r - r')^2]/s^2 = 2\,\mathrm{Var}_q(r)/s^2$, which is $2\sigma^2$ for
$s = 1$ and $2$ for $s = \sigma$. For $\ell_1$,
$\mathbb{E}|r - r'|/s = \sigma\,\mathbb{E}|u - u'|/s$. \qed

\paragraph{Proposition~\ref{prop:symm}.}
Let $\eta = (\epsilon_i - \epsilon_j)/s$. In case (a) the pair
$(\epsilon_i, \epsilon_j)$ is exchangeable, so $\eta$ and $-\eta$ have the same
law. In case (b), $-\eta = [(-\epsilon_i) - (-\epsilon_j)]/s$ has the same law
as $\eta$ because each $\epsilon$ is symmetric and they are independent.
Let $\Delta = [r(z_i) - r(z_j)]/s$ and $f(h) = \mathbb{E}\,d(h - \Delta - \eta)$.
Then $f(\Delta + v) = \mathbb{E}\,d(v - \eta) = \mathbb{E}\,d(v + \eta) =
\mathbb{E}\,d(-v - \eta) = f(\Delta - v)$, using first the symmetry of $\eta$
and then the evenness of $d$. A convex function symmetric about $\Delta$ is
minimized at $\Delta$:
$f(\Delta) \le \tfrac12 f(\Delta + v) + \tfrac12 f(\Delta - v) = f(\Delta+v)$,
with strict inequality if $d$ is strictly convex. For $d = |\cdot|$ the
minimizers are the medians of $\Delta + \eta$, a single point if $\eta$ has
positive density at $0$. For the pointwise contrast,
$\arg\min_h \mathbb{E}|h - r - \epsilon| = r + \mathrm{median}(\epsilon)$, and
$\mathrm{median}(\epsilon) \ne \mathbb{E}\epsilon = 0$ for skewed noise. \qed

\paragraph{Lemma~\ref{lem:bounded}.}
With $\bar x$ the group mean and $\hat s^2 = \sum_k (x_k - \bar x)^2/(G-1)$,
$(x_i - x_j)^2 \le 2(x_i - \bar x)^2 + 2(x_j - \bar x)^2
\le 2\sum_k (x_k - \bar x)^2 = 2(G-1)\hat s^2$. Equality holds when
$x_i - \bar x = -(x_j - \bar x)$ and every other $x_k = \bar x$. \qed

\paragraph{Proposition~\ref{prop:firstorder} and Corollary~\ref{cor:grpo}.}
With $\ell_i = \beta[\log\pol(z_i) - \log\polref(z_i)]$,
$g_i = \nabla_\theta\log\pol(z_i)$, and $\Delta_{ij} = \tilde r_i - \tilde r_j$,
the implemented loss (mean reduction over the $SP$ pair entries) has gradient
$\frac{\beta}{SP}\sum\sum_{i<j}\psi(e_{ij})(g_i - g_j)$ with
$e_{ij} = (\ell_i - \ell_j) - \Delta_{ij}$. At a window start
$e_{ij} = -\Delta_{ij}$, and $\psi$ is odd, so $\psi(e_{ij}) = -\psi(\Delta_{ij})$.
Regrouping by index,
$\sum_{i<j}\psi(\Delta_{ij})(g_i - g_j) = \sum_i g_i\sum_{j\ne i}\psi(\Delta_{ij})$,
since for $j < i$ the term $-\psi(\Delta_{ji})g_i = \psi(\Delta_{ij})g_i$. This
is Eq.~\eqref{eq:th-firstorder}.
For $d(e) = e^2$, $\psi(e) = 2e$ and
$\sum_{j\ne i}2(\tilde r_i - \tilde r_j) = 2G(\tilde r_i - \overline{\tilde r})$.
Because $\tilde r = r/\hat s$, this equals $2G(r_i - \bar r)/\hat s = 2GA_i$.
For $d = |\cdot|$, $\psi = \mathrm{sign}$ (with $\mathrm{sign}(0) = 0$, the
subgradient autograd uses), and
$\sum_j \mathrm{sign}(r_i - r_j) = (\mathrm{rank}_i - 1) - (G - \mathrm{rank}_i)$.
Oddness of $\psi$ gives $\sum_i u_i = 0$.
For the corollary, the GRPO loss with $\beta_{\mathrm{KL}}=0$ and ratio $\equiv 1$
is a mean over $SGL$ token entries, with gradient
$-\frac{1}{SGL}\sum A_i g_i$, using $\sum_t \nabla\log\pol(z_{i,t}) = g_i$.
The \rrebel{}-$\ell_2$-std gradient is $-\frac{2\beta G}{SP}\sum A_i g_i$, and
the ratio of the two is $2\beta G^2L/P = 4\beta GL/(G-1)$. Adam maps
$g \mapsto cg$ to $m\mapsto cm$, $v\mapsto c^2v$, so its step
$m/(\sqrt v + \epsilon)$ is unchanged up to $\epsilon$. \qed

\paragraph{Lemma~\ref{lem:cluster}.}
The point for a floor is $\frac1n\sum_x(\bar c(x,z^\star(x,\lmb)),
\bar s(x,z^\star(x,\lmb)))$, a function of the assignment
$x \mapsto z^\star(x,\lmb)$ alone. If the assignment does not depend on $x$,
there are at most $K$ assignments. If $z^\star$ maximizes a reward that is
continuous in $\lmb$, the maximizer can change between two floors only if the
order of two prompts' rewards changes between them, that is, only if some
difference $r_\lmb(x,z) - r_\lmb(x,z')$ changes sign. \qed

\paragraph{Proposition~\ref{prop:eps}.}
$\{\pi : C(\pi) \ge \tau\}$ is the intersection of the simplex with one
half-space, and $S$ is linear, so an optimum is attained at a vertex of this
polytope. Its vertices are point masses on prompts with $\bar c_z \ge \tau$ and
two-point mixtures lying on the hyperplane $C(\pi) = \tau$, which gives the
support bound. The set of attainable $(C(\pi), S(\pi))$ is the convex hull of
the points $(\bar c_z, \bar s_z)$, so $\max\{S : C = t\}$ is the upper concave
envelope $U(t)$, and $V(\tau) = \max_{t \ge \tau} U(t)$. Now let $\tau > c_0$
and suppose an optimal $\pi$ had $C(\pi) > \tau$. Let $z_0$ be a prompt of
maximal sentiment, so $\bar c_{z_0} \le c_0 < \tau$. If $S(\pi) <
\bar s_{z_0}$, the mixture $(1-\epsilon)\pi + \epsilon\delta_{z_0}$ is still
feasible for small $\epsilon$ and has larger $S$, a contradiction. Otherwise
$\pi$ is supported on maximal-sentiment prompts, so $C(\pi) \le c_0 < \tau$, also
a contradiction. Hence $C(\pi) = \tau$ and $V(\tau) = U(\tau)$. $U$ is
concave, so $V$ is concave and continuous on the interior of
$(c_0, c_{\max}]$. It is continuous at $c_{\max}$ because $U$ is upper
semicontinuous on a polytope. It is non-increasing because the feasible sets are
nested. For $\tau \le c_0$ the maximal-sentiment policy is feasible, so
$V(\tau) = U(c_0)$. \qed

\paragraph{Proposition~\ref{prop:penalty}.}
(i) $S$ is linear, $(\tau - C)_+^2$ is a convex function of a linear map, and
$\mathrm{KL}(\cdot\,\|\,\pi_0)$ is strictly convex, so the objective is
strictly concave and has a unique maximizer. The maximizer is interior,
because the KL term's derivative diverges at the boundary. Stationarity on the
simplex reads
$\bar s_z + \nu(\tau - C(\pi))_+\,\bar c_z - \beta(\log\frac{\pi_z}{\pi_{0,z}} + 1) =
\text{const}$, which is the Gibbs form with $\mu = \nu(\tau - C(\pi))_+$.
Along the Gibbs family, $\frac{d}{d\mu}C(\pi_\mu) =
\mathrm{Var}_{\pi_\mu}(\bar c)/\beta \ge 0$. Hence
$h(\mu) = \mu - \nu(\tau - C(\pi_\mu))_+$ is strictly increasing, with
$h(0) \le 0$ and $h(\mu) \to \infty$, so it has exactly one root $\mu_\nu$. At
the root, $(\tau - C)_+ = \mu_\nu/\nu$. If $C(\pi_0) \ge \tau$ then
$\mu_\nu = \mu^\star = 0$. Otherwise suppose $\mu_\nu > \mu^\star$: by
monotonicity $C(\pi_{\mu_\nu}) \ge C(\pi_{\mu^\star}) = \tau$, so
$\mu_\nu = \nu(\tau - C)_+ = 0$, a contradiction. Hence $\mu_\nu \le \mu^\star$.
(ii) The attainable pairs are the convex hull of the points, and $F$ depends
on $\pi$ only through $(C,S)$, so $\max F = \max_t\,[\,U(t) - \frac{\nu}{2}(\tau
- t)_+^2\,]$. This is the maximum of a concave function of one variable. For
$t > \tau$ it equals $U(t)$, which is non-increasing for $t \ge c_0$, so the
maximizer satisfies $t^\star \le \tau$ when $\tau > c_0$. The first-order
condition $0 \in \partial U(t^\star) + \nu(\tau - t^\star)$ gives the
shortfall. Realizing $U(t^\star)$ needs at most two prompts, as in
Proposition~\ref{prop:eps}.
(iii) $\partial F/\partial\pi_z = \bar s_z - \nu(\tau - C)_+\cdot(-\bar c_z)$. In
Eq.~\eqref{eq:reward-exp}, $\mu_g$ and $\tau$ are constant within a group, so
$r_i$ differs from $\bar s_{z_i} + \mu_g\bar c_{z_i}$ by the per-group constant
$-\mu_g\tau$. The implementation's division by $1 + \mu_g$ is a per-group
positive scale. Both are removed by standardization, by
Lemma~\ref{lem:invariance}. \qed

\section{Choosing the multiplier: a simulation}
\label{app:sim}

\texttt{analysis/constraint\_sim.py} simulates \rrebel{}-std on the real
statistics of the $48$ prompts that account for at least $98\%$ of the four
Reward~I policies' test-split choices. It uses their per-sentence mean content
and sentiment and their per-sample spread over $N=50$ task-model samples, on
five randomly chosen test sentences. Each window moves $\log\pi$ a
fraction (the \emph{step}) of the way to the window's target
$\pi\,e^{\tilde r/\beta}$, $\beta = 0.5$, with $\tilde r$ standardized within
groups of $G = 16$ prompts drawn from $\pi$. The full target corresponds to
step~$1$. The measured logit growth of the trained Reward~I policies
corresponds to a step of about $0.05$. Table~\ref{tab:sim} reports, over the
binding floors $\tau = 30$--$90$, the mean and worst distance of
$\mathbb{E}[c]$ from the floor, the number of distinct points among the $20$
floors, the hypervolume, and the largest standard deviation of $\mathbb{E}[c]$
over the last $20$ windows (``late wobble''). The proportional rule, with
$\nu = 5$, is the rule used in our experiments.

\section{Numerical verification of the theory}
\label{app:checks}

\texttt{analysis/theory\_checks.py} tests each statement of
Section~\ref{sec:theory} on a tabular single-token policy (so that
$\log\pol(z)$ is available in closed form). Checks 1--3 call the repository's
own \texttt{\_rrebel\_core} and \texttt{grpo\_v2\_loss}, so they verify the
\emph{implementation}; the rest compute the idealized quantities exactly. The
script exits non-zero if any check fails. All $40$ checks pass. The exact
identities are run with the code's $\varepsilon = 10^{-8}$ standard-deviation
guard set to zero: with the guard in place, the invariance and gradient-ratio
checks deviate by $\sim\!10^{-7}$, which is exactly the size of $\varepsilon/\sigma$
on those toy scales.
